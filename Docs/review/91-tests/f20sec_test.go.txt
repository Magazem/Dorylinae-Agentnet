// Review 91 (R55-F20 security): cost of idTakenTx, which runs before the
// daily cap, on a large requests table.
func TestF20SecIDTakenScanCost(t *testing.T) {
	s, _, _ := newTestStore(t, testTo, &policy{})
	fakeRow(t, s, "in", testKey(3), NewID(), "seed")
	const n = 100000
	cols, err := s.DB.Query(`SELECT name FROM pragma_table_info('requests')`)
	if err != nil {
		t.Fatal(err)
	}
	var sel []string
	for cols.Next() {
		var c string
		_ = cols.Scan(&c)
		switch c {
		case "peer":
			sel = append(sel, "'peer' || (c.x % 50)")
		case "id":
			sel = append(sel, "printf('r-%032x', c.x)")
		case "idem_key":
			sel = append(sel, "NULL")
		default:
			sel = append(sel, "r."+c)
		}
	}
	cols.Close()
	if _, err := s.DB.Exec(`WITH RECURSIVE c(x) AS (SELECT 1 UNION ALL SELECT x+1 FROM c WHERE x < ?)
INSERT INTO requests SELECT `+strings.Join(sel, ", ")+` FROM c, (SELECT * FROM requests LIMIT 1) r`, n); err != nil {
		t.Fatal(err)
	}
	rows, err := s.DB.Query(`EXPLAIN QUERY PLAN SELECT 1 FROM requests WHERE id = ? AND NOT (direction = 'in' AND peer = ?) LIMIT 1`, "r-x", "p")
	if err != nil {
		t.Fatal(err)
	}
	for rows.Next() {
		var a, b, c int
		var d string
		_ = rows.Scan(&a, &b, &c, &d)
		t.Logf("plan: %s", d)
	}
	rows.Close()
	ctx := context.Background()
	const iters = 50
	start := time.Now()
	for i := 0; i < iters; i++ {
		tx, _ := s.DB.BeginTx(ctx, nil)
		if _, err := idTakenTx(ctx, tx, NewID(), testFrom); err != nil {
			t.Fatal(err)
		}
		_ = tx.Rollback()
	}
	t.Logf("rows=%d idTakenTx avg=%v", n, time.Since(start)/iters)
}

// Review 91 (R55-F20 security): the R55-163 {} row from an unverified peer
// still carries that peer's cancel reason text, which request_show returns.
func TestF20SecTombstonedRowKeepsPeerReason(t *testing.T) {
	pol := policy{unverified: true}
	s, _, _ := newTestStore(t, testTo, &pol)
	now := time.Now()
	req := receivedRequest()
	if err := deliverMirror(t, s, KindCancel, testFrom, now, map[string]any{"at": wireTime(now), "request": req.ID, "reason": "PEER TEXT visit evil.example"}); err != nil {
		t.Fatal(err)
	}
	if err := deliverRequest(t, s, req, now.Add(time.Second)); err != nil {
		t.Fatal(err)
	}
	v, err := s.Show(context.Background(), req.ID, testFrom)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("title=%q reason=%q", v.Title, v.Reason)
	if v.Reason != "" {
		t.Errorf("unverified peer's cancel reason kept on the {} row: %q", v.Reason)
	}
}
