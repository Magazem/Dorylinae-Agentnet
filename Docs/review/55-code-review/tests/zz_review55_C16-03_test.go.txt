package logfile

import (
	"os"
	"path/filepath"
	"testing"
)

// Review 55 C16-03: a failed rotation (here: another process holds the log
// open without FILE_SHARE_DELETE, as a Windows log viewer may) leaves the
// writer closed for the rest of the daemon's life.
func TestReview55C16_03RotateFailureKillsLog(t *testing.T) {
	p := filepath.Join(t.TempDir(), "d.log")
	w, err := Open(p, 10)
	if err != nil {
		t.Fatal(err)
	}
	defer func() { _ = w.Close() }()
	if _, err := w.Write([]byte("0123456789")); err != nil {
		t.Fatal(err)
	}
	viewer, err := os.Open(p)
	if err != nil {
		t.Fatal(err)
	}
	_, err1 := w.Write([]byte("rotate-now\n"))
	_ = viewer.Close()
	_, err2 := w.Write([]byte("after viewer closed\n"))
	t.Logf("write during viewer: %v; write after viewer closed: %v", err1, err2)
	if err1 != nil && err2 != nil {
		t.Errorf("log writer stays dead after one failed rotation: %v", err2)
	}
}
