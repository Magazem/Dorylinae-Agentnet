package daemon

import (
	"strings"
	"testing"

	"github.com/Magazem/Dorylinae-Agentnet/internal/notify"
)

// Review 55 T11-01: a peer name whose code-like digits are split by
// invisible characters survives Clean + stripLongDigits (device link,
// device scope and debate constraint summaries).
func TestReview55T11_01(t *testing.T) {
	for _, name := range []string{
		"Code 4​8​2​9​1​3",  // zero-width space (Cf)
		"Code 4⁠8⁠2⁠9⁠1⁠3",  // word joiner (Cf)
		"Code 4️8️2️9️1️3",  // variation selector (Mn)
		"Code 4ㅤ8ㅤ2ㅤ9ㅤ1ㅤ3",  // Hangul filler (Lo)
		`Code \064\070\062\071\061\063`,              // octal escapes (zenity g_strcompress)
		"Code 482913",                                  // control: stripped
	} {
		got := stripLongDigits(notify.Clean(name, 40))
		t.Logf("%q -> %q (stripped=%v)", name, got, strings.Contains(got, "…"))
	}
}
