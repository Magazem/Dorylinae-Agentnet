package main

import (
	"testing"

	"github.com/Magazem/Dorylinae-Agentnet/internal/decision"
)

// Review 55 T11-04: termSafe (unicode.IsPrint) lets through the review 46 H1
// "graphic but invisible" set that decision.Visible escapes.
func TestReview55T11_04(t *testing.T) {
	for _, s := range []string{
		"reportㅤ.txt", // Hangul filler (Lo)
		"reportᅟ.txt", // Hangul choseong filler (Lo)
		"report️.txt", // variation selector (Mn)
		"report⠀.txt", // braille blank (So)
		"report​.txt", // zero-width space (Cf): quoted
		"report‮.txt", // RLO (Cf): quoted
	} {
		t.Logf("%q: termSafe=%q Visible=%q", s, termSafe(s), decision.Visible(s, false))
	}
}
