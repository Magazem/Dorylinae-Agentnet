package relay_test

import (
	"fmt"
	"testing"
	"time"

	"github.com/coder/websocket"

	"github.com/Magazem/Dorylinae-Agentnet/internal/relay"
)

// Review 55 verification of C01-02, at the early relay's real size
// (--max-inflight 48MiB) with every other limit at its default, from ONE /24.
// Each attacker connection sends a little over half a frame's worth and stops.
func TestReview55C01_02VerifyOnePrefixFullSize(t *testing.T) {
	const budget = 48 << 20
	e := newLimitEnv(t, relay.Options{MaxInflight: budget})
	victim, other := newPeer(t), newPeer(t)
	cv := e.authed(victim, "10.9.0.1")
	e.authed(other, "10.9.0.2")

	start := time.Now()
	sent, n := 0, 0
	for ; n < 63 && e.s.Reading() > budget-4096 == false; n++ {
		left := int(budget - e.s.Reading())
		w := 1<<20 - 1 // just under a whole frame: charges the full ~1 MiB buffer
		switch {
		case left < 64<<10:
			w = 5000 // flushed by the client library; charges one 8 KiB buffer
		case left < 1200<<10:
			w = left / 3 // a buffer that surely still fits
		}
		a := newPeer(t)
		ca := e.authed(a, fmt.Sprintf("10.1.0.%d", n+1)) // all in 10.1.0.0/24
		wr, err := ca.Writer(ctx(t), websocket.MessageText)
		if err != nil {
			t.Fatal(err)
		}
		if _, err := wr.Write(payload(w)); err != nil {
			t.Fatal(err)
		}
		sent += w
		time.Sleep(30 * time.Millisecond)
		t.Logf("conn %d wrote %d, reading=%d", n, w, e.s.Reading())
	}
	t.Logf("one /24: %d connections, %d KiB uploaded, hold %d of %d read-budget bytes (%s)",
		n, sent>>10, e.s.Reading(), budget, time.Since(start).Round(time.Millisecond))
	if e.s.Reading() <= budget-4096 {
		t.Fatalf("attacker holds only %d of %d", e.s.Reading(), budget)
	}
	// Advance past any per-minute window: nothing times the unfinished frames out.
	time.Sleep(2 * time.Second)
	writeFrame(t, cv, []byte(mustJSON(victim.env(other.key, "honest-1", payload(200)))))
	expectClose(t, cv, websocket.StatusTryAgainLater)
	t.Logf("still held after 2s: %d bytes; honest 200-byte mail closed 1013", e.s.Reading())
}
