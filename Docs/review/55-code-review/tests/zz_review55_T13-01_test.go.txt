//go:build darwin

package daemon

import (
	"context"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/Magazem/Dorylinae-Agentnet/internal/capability"
)

// Review 55 T13-01: /Users is a firmlink to /System/Volumes/Data/Users
// (macOS 10.15+). EvalSymlinks does not resolve firmlinks, so the Data-volume
// spelling of a forbidden resource may pass issuance step 3.
//
// Run on a macOS box or runner:
//   go test ./internal/daemon -run 'TestReview55T1301' -count=1 -v
func TestReview55T1301FirmlinkForbidden(t *testing.T) {
	const data = "/System/Volumes/Data"
	home, err := os.UserHomeDir()
	if err != nil || !strings.HasPrefix(home, "/Users/") {
		t.Skipf("home %q is not under /Users", home)
	}
	if _, err := os.Stat(data + home); err != nil {
		t.Skipf("no firmlinked data volume: %v", err)
	}
	// A config dir under the real home, like the default
	// ~/Library/Application Support/dorylinae.
	cfg, err := os.MkdirTemp(home, "zz-review55-T13-01-")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.RemoveAll(cfg) })
	if err := os.WriteFile(filepath.Join(cfg, "dorylinae.db"), []byte("SECRET-DB"), 0o600); err != nil {
		t.Fatal(err)
	}

	// Precondition: both spellings are the same directory.
	a, _ := os.Stat(home)
	b, _ := os.Stat(data + home)
	if !os.SameFile(a, b) {
		t.Fatalf("%s and %s are not the same directory", home, data+home)
	}
	// Control: the normal spellings are refused.
	for _, p := range []string{home, cfg, "/"} {
		if _, ierr := validateResource(cfg, p); ierr == nil {
			t.Fatalf("control: %q accepted", p)
		}
	}

	bypass := false
	for _, p := range []string{data + cfg, data + home, data, data + "/Users"} {
		resolved, ierr := validateResource(cfg, p)
		if ierr != nil {
			t.Logf("refused %q: %s", p, ierr.Message)
			continue
		}
		bypass = true
		t.Errorf("ACCEPTED %q -> %q", p, resolved)
	}

	// Serving: the config dir under its Data spelling hands out the DB.
	if resolved, ierr := validateResource(cfg, data+cfg); ierr == nil {
		var got []byte
		err := capability.FSBackend{}.Read(context.Background(),
			capability.Record{Path: resolved, Action: capability.ActionFSRead}, "dorylinae.db", 0, 1024,
			func(f capability.Fragment) error { got = append(got, f.Data...); return nil })
		if err == nil && string(got) == "SECRET-DB" {
			t.Errorf("config dir served through %q", resolved)
		}
		// recheckResource (approval Precondition) must also refuse it.
		if rerr := recheckResource(cfg, capability.Record{Path: resolved, Action: capability.ActionFSRead}); rerr == nil {
			t.Errorf("recheckResource accepted %q", resolved)
		}
	}
	if !bypass {
		t.Log("no bypass: finding T13-01 does not reproduce on this macOS")
	}
}
