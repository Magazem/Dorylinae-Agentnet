package mail

import (
	"context"
	"database/sql"
	"sync"
	"testing"
	"time"
)

// Review 55 T9-01: a paired peer reuses the id of one of its own mails after
// mail_seen was pruned (35 d) but mail_inbox (never pruned) still holds it.
// Apply runs and succeeds, then the mail_inbox INSERT fails on its primary key,
// so every delivery rolls back, is never acked, and leaves whatever Apply put in
// a pending* map (keyed by *Opened) behind, because After never runs.
func TestReview55T9_01(t *testing.T) {
	f := newRecvFixture(t)
	var applies int
	var pending sync.Map // the pending* pattern of the real kinds
	var afters int
	f.rcv.Kinds["request"] = Kind{
		Inbox: true,
		Apply: func(_ context.Context, _ *sql.Tx, op *Opened) error {
			applies++
			pending.Store(op, struct{}{})
			return nil
		},
		After: func(_ context.Context, op *Opened) { afters++; pending.Delete(op) },
	}
	ctx := context.Background()
	if err := f.rcv.Handle(ctx, f.mailEnv(testID, "request", map[string]any{"text": "first"})); err != nil {
		t.Fatalf("first delivery: %v", err)
	}
	// 36 days later: mail_seen is pruned, mail_inbox is not.
	later := vectorNow.Add(36 * 24 * time.Hour)
	if _, err := Prune(ctx, f.st.DB(), later); err != nil {
		t.Fatal(err)
	}
	f.rcv.Now = func() time.Time { return later }
	f.rcv.Opener.Now = func() time.Time { return later }
	acksBefore := f.out.count()
	for i := 0; i < 5; i++ {
		sl := sealTo(t, f.sender, f.recip, testID, "request", map[string]any{"text": "again"}, later)
		err := f.rcv.Handle(ctx, env(f.sender, f.recip, sl))
		if i == 0 {
			t.Logf("re-used id delivery error: %v", err)
		}
	}
	n := 0
	pending.Range(func(_, _ any) bool { n++; return true })
	t.Logf("applies=%d afters=%d leaked pending entries=%d acks sent for re-used id=%d mail_seen rows=%d",
		applies, afters, n, f.out.count()-acksBefore, f.count(`SELECT COUNT(*) FROM mail_seen`))
	if n > 0 && f.out.count() == acksBefore {
		t.Fatalf("T9-01: re-used mail id is applied %d times, never acked, and leaks %d pending entries", applies-1, n)
	}
}
