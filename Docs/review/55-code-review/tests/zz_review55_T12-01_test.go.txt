package mailbox_test

// Review 55 theme T12, finding T12-01: a forward wall-clock step (then
// corrected) deletes every mailbox key peers know and leaves a current key
// whose announcement peers refuse until real time reaches its created.

import (
	"context"
	"encoding/base64"
	"errors"
	"testing"
	"time"

	"github.com/Magazem/Dorylinae-Agentnet/internal/mail"
)

func TestReview55T12_01ForwardClockStep(t *testing.T) {
	dir := t.TempDir()
	db := newDB(t)
	t0 := time.Date(2026, 5, 1, 12, 0, 0, 0, time.UTC)
	now := t0
	k, pub, _ := newKeysOn(t, dir, db, func() time.Time { return now })
	identity := base64.RawURLEncoding.EncodeToString(pub)

	first, err := k.Announcement()
	if err != nil {
		t.Fatal(err)
	}
	firstID := keyIDOf(t, first, identity, now)

	// The clock jumps 30 days ahead for one hourly Rotate run.
	now = t0.Add(30 * 24 * time.Hour)
	if _, err := k.Rotate(context.Background()); err != nil {
		t.Fatal(err)
	}
	// NTP corrects it an hour later (true time).
	now = t0.Add(2 * time.Hour)
	if _, err := k.Rotate(context.Background()); err != nil {
		t.Fatal(err)
	}

	if _, live := k.MailboxKey(firstID); live {
		t.Fatalf("first key still live")
	}
	t.Logf("T12-01: key %s that peers seal to was deleted during the step", firstID)

	cur, err := k.Announcement()
	if err != nil {
		t.Fatal(err)
	}
	_, _, perr := mail.ParseAnnouncement(cur, identity, now)
	if !errors.Is(perr, mail.ErrAnnouncementExpired) {
		t.Fatalf("peer would accept the current announcement (err=%v)", perr)
	}
	t.Logf("T12-01: current announcement refused by a peer at true time: %v", perr)

	// Still stuck 20 days later (true time), well before the future created.
	now = t0.Add(20 * 24 * time.Hour)
	if _, err := k.Rotate(context.Background()); err != nil {
		t.Fatal(err)
	}
	cur, _ = k.Announcement()
	if _, _, perr := mail.ParseAnnouncement(cur, identity, now); errors.Is(perr, mail.ErrAnnouncementExpired) {
		t.Logf("T12-01: still refused 20 days later: no usable mailbox key for peers")
	}
}
